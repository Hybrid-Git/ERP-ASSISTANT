# vLLM Server Startup Command for Runpod (Qwen 14B + Tool Calling)
# Run this in your Runpod terminal where the model is stored.

python -m vllm.entrypoints.openai.api_server \
    --model Qwen/Qwen3-14B-AWQ \
    --enable-auto-tool-choice \
    --tool-call-parser qwen \
    --max-model-len 16384 \
    --gpu-memory-utilization 0.9 \
    --host 0.0.0.0 \
    --port 8000
