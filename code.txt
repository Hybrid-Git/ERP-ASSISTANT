        try:
            with requests.post(
                API_URL,
                json={"query": prompt, "session_id": st.session_state.session_id},
                headers=_auth_headers,
                stream=True,
                timeout=180,
            ) as resp:
                resp.raise_for_status()

                buf = ""
                for raw in resp.iter_content(chunk_size=1, decode_unicode=True):
                    buf += raw
                    while "\n" in buf:
                        line, buf = buf.split("\n", 1)
                        line = line.strip()
                        if not line or not line.startswith("data: "):
                            continue
                        payload = json.loads(line[len("data: "):])

                        if "token" in payload:
                            if not tokens_started:
                                tokens_started = True
                                status_ph.empty()
                            full_response += payload["token"]
                            if full_response.strip():
                                response_ph.markdown(full_response + "\u258c")

                        elif "error" in payload:
                            error_msg = payload["error"].get("message") if isinstance(payload["error"], dict) else str(payload["error"])

                        elif payload.get("done"):
                            break

        except requests.exceptions.ConnectionError:
            error_msg = f"Backend not reachable \u2014 is the FastAPI server running at {_backend_base}?"
        except requests.exceptions.Timeout:
            error_msg = "The request timed out. Please try again."
        except requests.exceptions.RequestException as e:
            error_msg = str(e)
